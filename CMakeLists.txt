cmake_minimum_required(VERSION 3.15)
project(ProxyServer VERSION 2.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_library(proxy_config STATIC src/gost_config.cpp)
target_include_directories(proxy_config PUBLIC src)
if(WIN32)
  enable_language(RC)
  find_package(Threads REQUIRED)
  add_executable(ProxyServer WIN32 src/main_win.cpp src/proxy_backend_win.cpp src/child_process_win.cpp src/mtproto_win.cpp src/service_host_win.cpp src/service_codec.cpp src/service_firewall_plan.cpp src/service_firewall_win.cpp resources/app.rc)
  target_include_directories(ProxyServer PRIVATE src resources)
  target_compile_definitions(ProxyServer PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(ProxyServer PRIVATE proxy_config Threads::Threads ws2_32 crypt32 advapi32 user32 shell32 gdi32 comctl32 comdlg32 bcrypt winhttp ole32 oleaut32 uuid)
  if(MSVC)
    target_link_options(ProxyServer PRIVATE /MANIFEST:NO)
  elseif(MINGW)
    target_link_options(ProxyServer PRIVATE -municode -static -static-libgcc -static-libstdc++)
  endif()
  add_custom_command(TARGET ProxyServer POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/components" "$<TARGET_FILE_DIR:ProxyServer>/components")
endif()
option(PROXY_BUILD_TESTS "Build the portable configuration and service tests" OFF)
if(PROXY_BUILD_TESTS OR NOT WIN32)
  add_executable(proxy_config_test tests/config_cli.cpp)
  target_link_libraries(proxy_config_test PRIVATE proxy_config)
  add_executable(service_firewall_test tests/test_service_firewall.cpp src/service_firewall_plan.cpp)
  target_include_directories(service_firewall_test PRIVATE src)
  add_executable(service_codec_test tests/test_service_codec.cpp src/service_codec.cpp)
  target_include_directories(service_codec_test PRIVATE src)
  enable_testing()
  add_test(NAME service_firewall_plan COMMAND service_firewall_test)
  add_test(NAME service_codec COMMAND service_codec_test)
endif()
